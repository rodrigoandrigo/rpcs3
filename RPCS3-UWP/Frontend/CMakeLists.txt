cmake_minimum_required(VERSION 3.21)

set(CMAKE_SYSTEM_NAME "WindowsStore" CACHE STRING "" FORCE)
set(CMAKE_SYSTEM_VERSION "10.0" CACHE STRING "" FORCE)
set(CMAKE_VS_WINDOWS_TARGET_PLATFORM_MIN_VERSION "10.0.19041.0"
	CACHE STRING "" FORCE)
set(VCPKG_TARGET_TRIPLET "x64-uwp-static" CACHE STRING "" FORCE)
set(VCPKG_OVERLAY_TRIPLETS "${CMAKE_CURRENT_SOURCE_DIR}/cmake/triplets"
	CACHE STRING "" FORCE)

if(NOT DEFINED CMAKE_TOOLCHAIN_FILE AND DEFINED ENV{VCPKG_ROOT})
	set(CMAKE_TOOLCHAIN_FILE
		"$ENV{VCPKG_ROOT}/scripts/buildsystems/vcpkg.cmake"
		CACHE FILEPATH "vcpkg toolchain file")
endif()

project(UwpImGuiFrontendKit LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(UWP_IMGUI_FRONTEND_IMGUI_DIR
	"${CMAKE_CURRENT_SOURCE_DIR}/dependencies/imgui"
	CACHE PATH "Directory containing imgui.h")
set(UWP_IMGUI_FRONTEND_BUILD_SAMPLE OFF CACHE BOOL "" FORCE)
option(UWP_IMGUI_FRONTEND_KIT_BUILD_TESTS "Build frontend host-logic tests" OFF)
set(UWP_IMGUI_FRONTEND_BUILD_TESTS
	${UWP_IMGUI_FRONTEND_KIT_BUILD_TESTS} CACHE BOOL "" FORCE)

add_subdirectory(UWP-ImGuiFrontend)
add_subdirectory(../FrontendHost "${CMAKE_CURRENT_BINARY_DIR}/RPCS3-UWP-Host")

set_property(DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" PROPERTY
	VS_STARTUP_PROJECT RPCS3UwpApp)
